18/// Configuration options for JSON Schema validation.
19#[derive(Clone)]
20pub struct ValidationOptions<R = Arc<dyn Retrieve>> {
21    pub(crate) draft: Option<Draft>,
22    content_media_type_checks: AHashMap<&'static str, Option<ContentMediaTypeCheckType>>,
23    content_encoding_checks_and_converters:
24        AHashMap<&'static str, Option<(ContentEncodingCheckType, ContentEncodingConverterType)>>,
25    /// Retriever for external resources
26    pub(crate) retriever: R,
27    /// Additional resources that should be addressable during validation.
28    pub(crate) resources: AHashMap<String, Resource>,
29    pub(crate) registry: Option<referencing::Registry>,
30    formats: AHashMap<String, Arc<dyn Format>>,
31    validate_formats: Option<bool>,
32    pub(crate) validate_schema: bool,
33    ignore_unknown_formats: bool,
34    keywords: AHashMap<String, Arc<dyn KeywordFactory>>,
35}

168    /// Add support for a custom content encoding.
169    ///
170    /// # Arguments
171    ///
172    /// * `encoding`: Name of the content encoding (e.g., "base64")
173    /// * `check`: Validates the input string (return `true` if valid)
174    /// * `converter`: Converts the input string, returning:
175    ///   - `Err(ValidationError)`: For supported errors
176    ///   - `Ok(None)`: If input is invalid
177    ///   - `Ok(Some(content))`: If valid, with decoded content
178    ///
179    /// # Example
180    ///
181    /// ```rust
182    /// use jsonschema::ValidationError;
183    ///
184    /// fn check(s: &str) -> bool {
185    ///     s.starts_with("valid:")
186    /// }
187    ///
188    /// fn convert(s: &str) -> Result<Option<String>, ValidationError<'static>> {
189    ///     if s.starts_with("valid:") {
190    ///         Ok(Some(s[6..].to_string()))
191    ///     } else {
192    ///         Ok(None)
193    ///     }
194    /// }
195    ///
196    /// let options = jsonschema::options()
197    ///     .with_content_encoding("custom", check, convert);
198    /// ```

248    ///
249    /// This enables the use of additional in-memory schemas alongside the main schema being validated.
250    ///
251    /// # Example
252    ///
253    /// ```rust
254    /// # use serde_json::json;

381    /// Register a custom keyword validator.
382    ///
383    /// ## Example
384    ///
385    /// ```rust
386    /// # use jsonschema::{
387    /// #    paths::{LazyLocation, Location},
388    /// #    ErrorIterator, Keyword, ValidationError,
389    /// # };
390    /// # use serde_json::{json, Map, Value};
391    /// # use std::iter::once;
392    ///
393    /// struct MyCustomValidator;
394    ///
395    /// impl Keyword for MyCustomValidator {
396    ///     fn validate<'i>(
397    ///         &self,
398    ///         instance: &'i Value,
399    ///         location: &LazyLocation,
400    ///     ) -> Result<(), ValidationError<'i>> {
401    ///         // ... validate instance ...
402    ///         if !instance.is_object() {
403    ///             return Err(ValidationError::custom(
404    ///                 Location::new(),
405    ///                 location.into(),
406    ///                 instance,
407    ///                 "Boom!",
408    ///             ));
409    ///         } else {
410    ///             Ok(())
411    ///         }
412    ///     }
413    ///     fn is_valid(&self, instance: &Value) -> bool {
414    ///         // ... determine if instance is valid ...
415    ///         true
416    ///     }
417    /// }
418    ///
419    /// // You can create a factory function, or use a closure to create new validator instances.
420    /// fn custom_validator_factory<'a>(
421    ///     parent: &'a Map<String, Value>,
422    ///     value: &'a Value,
423    ///     path: Location,
424    /// ) -> Result<Box<dyn Keyword>, ValidationError<'a>> {
425    ///     Ok(Box::new(MyCustomValidator))
426    /// }
427    ///
428    /// let validator = jsonschema::options()
429    ///     .with_keyword("my-type", custom_validator_factory)
430    ///     .with_keyword("my-type-with-closure", |_, _, _| Ok(Box::new(MyCustomValidator)))
431    ///     .build(&json!({ "my-type": "my-schema"}))
432    ///     .expect("A valid schema");
433    ///
434    /// assert!(validator.is_valid(&json!({ "a": "b"})));
435    /// ```
436    pub fn with_keyword<N, F>(mut self, name: N, factory: F) -> Self
437    where
438        N: Into<String>,
439        F: for<'a> Fn(
440                &'a serde_json::Map<String, Value>,
441                &'a Value,
442                Location,
443            ) -> Result<Box<dyn Keyword>, ValidationError<'a>>
444            + Send
445            + Sync
446            + 'static,
447    {
448        self.keywords.insert(name.into(), Arc::new(factory));
449        self
450    }

458    /// Build a JSON Schema validator using the current options.
459    ///
460    /// # Example
461    ///
462    /// ```rust
463    /// use serde_json::json;
464    ///
465    /// let schema = json!({"type": "string"});
466    /// let validator = jsonschema::options()
467    ///     .build(&schema)
468    ///     .expect("A valid schema");
469    ///
470    /// assert!(validator.is_valid(&json!("Hello")));
471    /// assert!(!validator.is_valid(&json!(42)));
472    /// ```
473    pub fn build(&self, schema: &Value) -> Result<Validator, ValidationError<'static>> {
474        compiler::build_validator(self.clone(), schema)
475    }
476    pub(crate) fn draft_for(&self, contents: &Value) -> Result<Draft, ValidationError<'static>> {
477        // Preference:
478        //  - Explicitly set
479        //  - Autodetected
480        //  - Default
481        if let Some(draft) = self.draft {
482            Ok(draft)
483        } else {
484            let default = Draft::default();
485            match default.detect(contents) {
486                Ok(draft) => Ok(draft),
487                Err(referencing::Error::UnknownSpecification { specification }) => {
488                    // Try to retrieve the specification and detect its draft
489                    if let Ok(Ok(retrieved)) =
490                        uri::from_str(&specification).map(|uri| self.retriever.retrieve(&uri))
491                    {
492                        Ok(default.detect(&retrieved)?)
493                    } else {
494                        Err(referencing::Error::UnknownSpecification { specification }.into())
495                    }
496                }
497                Err(error) => Err(error.into()),
498            }
499        }
500    }
501    /// Set a retriever to fetch external resources.
502    pub fn with_retriever(mut self, retriever: impl Retrieve + 'static) -> Self {
503        self.retriever = Arc::new(retriever);
504        self
505    }

531    pub(crate) async fn draft_for(
532        &self,
533        contents: &Value,
534    ) -> Result<Draft, ValidationError<'static>> {
535        // Preference:
536        //  - Explicitly set
537        //  - Autodetected
538        //  - Default
539        if let Some(draft) = self.draft {
540            Ok(draft)
541        } else {
542            let default = Draft::default();
543            match default.detect(contents) {
544                Ok(draft) => Ok(draft),
545                Err(referencing::Error::UnknownSpecification { specification }) => {
546                    // Try to retrieve the specification and detect its draft
547                    if let Ok(uri) = uri::from_str(&specification) {
548                        if let Ok(retrieved) = self.retriever.retrieve(&uri).await {
549                            return Ok(default.detect(&retrieved)?);
550                        }
551                    }
552                    Err(referencing::Error::UnknownSpecification { specification }.into())
553                }
554                Err(error) => Err(error.into()),
555            }
556        }
557    }
