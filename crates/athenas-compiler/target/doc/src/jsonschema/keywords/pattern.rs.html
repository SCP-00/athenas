15static REGEX_CACHE: Lazy<Mutex<LruCache>> = Lazy::new(|| Mutex::new(LruCache::new(10)));
16
17struct LruCache {
18    map: AHashMap<String, fancy_regex::Regex>,
19    queue: VecDeque<String>,
20    capacity: usize,
21}
22
23impl LruCache {
24    fn new(capacity: usize) -> Self {
25        LruCache {
26            map: AHashMap::new(),
27            queue: VecDeque::new(),
28            capacity,
29        }
30    }
31
32    fn get(&mut self, key: &str) -> Option<&fancy_regex::Regex> {
33        if let Some(value) = self.map.get(key) {
34            let index = self.queue.iter().position(|x| x == key).unwrap();
35            let k = self.queue.remove(index).unwrap();
36            self.queue.push_back(k);
37            Some(value)
38        } else {
39            None
40        }
41    }
42
43    fn insert(&mut self, key: String, value: fancy_regex::Regex) -> Option<fancy_regex::Regex> {
44        if self.map.len() >= self.capacity && !self.map.contains_key(&key) {
45            if let Some(lru_key) = self.queue.pop_front() {
46                self.map.remove(&lru_key);
47            }
48        }
49
50        let old_value = self.map.insert(key.clone(), value);
51        if old_value.is_some() {
52            let index = self.queue.iter().position(|x| x == &key).unwrap();
53            self.queue.remove(index);
54        }
55        self.queue.push_back(key);
56        old_value
57    }
58}

60pub(crate) struct PatternValidator {
61    original: String,
62    pattern: fancy_regex::Regex,
63    location: Location,
64}
65
66impl PatternValidator {
67    #[inline]
68    pub(crate) fn compile<'a>(
69        ctx: &compiler::Context,
70        pattern: &'a Value,
71    ) -> CompilationResult<'a> {
72        match pattern {
73            Value::String(item) => {
74                let mut cache = REGEX_CACHE.lock().expect("Lock is poisoned");
75                let pattern = if let Some(regex) = cache.get(item) {
76                    regex.clone()
77                } else {
78                    let regex = match ecma::to_rust_regex(item)
79                        .map(|pattern| fancy_regex::Regex::new(&pattern))
80                    {
81                        Ok(Ok(r)) => r,
82                        _ => {
83                            return Err(ValidationError::format(
84                                Location::new(),
85                                ctx.location().clone(),
86                                pattern,
87                                "regex",
88                            ))
89                        }
90                    };
91                    cache.insert(item.clone(), regex.clone());
92                    regex
93                };
94                Ok(Box::new(PatternValidator {
95                    original: item.clone(),
96                    pattern,
97                    location: ctx.location().join("pattern"),
98                }))
99            }
100            _ => Err(ValidationError::single_type_error(
101                Location::new(),
102                ctx.location().clone(),
103                pattern,
104                PrimitiveType::String,
105            )),
106        }
107    }
108}

110impl Validate for PatternValidator {
111    fn validate<'i>(
112        &self,
113        instance: &'i Value,
114        location: &LazyLocation,
115    ) -> Result<(), ValidationError<'i>> {
116        if let Value::String(item) = instance {
117            match self.pattern.is_match(item) {
118                Ok(is_match) => {
119                    if !is_match {
120                        return Err(ValidationError::pattern(
121                            self.location.clone(),
122                            location.into(),
123                            instance,
124                            self.original.clone(),
125                        ));
126                    }
127                }
128                Err(e) => {
129                    return Err(ValidationError::backtrack_limit(
130                        self.location.clone(),
131                        location.into(),
132                        instance,
133                        e,
134                    ));
135                }
136            }
137        }
138        Ok(())
139    }
140
141    fn is_valid(&self, instance: &Value) -> bool {
142        if let Value::String(item) = instance {
143            return self.pattern.is_match(item).unwrap_or(false);
144        }
145        true
146    }
147}

149#[inline]
150pub(crate) fn compile<'a>(
151    ctx: &compiler::Context,
152    _: &'a Map<String, Value>,
153    schema: &'a Value,
154) -> Option<CompilationResult<'a>> {
155    Some(PatternValidator::compile(ctx, schema))
156}
