177pub(crate) fn get_for_draft<'a>(
178    ctx: &compiler::Context<'a>,
179    keyword: &'a str,
180) -> Option<(Keyword, CompileFunc<'a>)> {
181    match (ctx.draft(), keyword) {
182        // Keywords common to all drafts
183        (_, "$ref") => Some((BuiltinKeyword::Ref.into(), ref_::compile_ref)),
184        (_, "additionalItems") if ctx.has_vocabulary(&Vocabulary::Applicator) => Some((
185            BuiltinKeyword::AdditionalItems.into(),
186            additional_items::compile,
187        )),
188        (_, "additionalProperties") if ctx.has_vocabulary(&Vocabulary::Applicator) => Some((
189            BuiltinKeyword::AdditionalProperties.into(),
190            additional_properties::compile,
191        )),
192        (_, "allOf") if ctx.has_vocabulary(&Vocabulary::Applicator) => {
193            Some((BuiltinKeyword::AllOf.into(), all_of::compile))
194        }
195        (_, "anyOf") if ctx.has_vocabulary(&Vocabulary::Applicator) => {
196            Some((BuiltinKeyword::AnyOf.into(), any_of::compile))
197        }
198        (_, "dependencies") if ctx.has_vocabulary(&Vocabulary::Applicator) => {
199            Some((BuiltinKeyword::Dependencies.into(), dependencies::compile))
200        }
201        (_, "enum") if ctx.has_vocabulary(&Vocabulary::Validation) => {
202            Some((BuiltinKeyword::Enum.into(), enum_::compile))
203        }
204        (Draft::Draft201909, "format") if ctx.has_vocabulary(&Vocabulary::Format) => {
205            Some((BuiltinKeyword::Format.into(), format::compile))
206        }
207        (Draft::Draft202012, "format") if ctx.has_vocabulary(&Vocabulary::FormatAnnotation) => {
208            Some((BuiltinKeyword::Format.into(), format::compile))
209        }
210        (_, "format") => Some((BuiltinKeyword::Format.into(), format::compile)),
211        (_, "items") if ctx.has_vocabulary(&Vocabulary::Applicator) => {
212            Some((BuiltinKeyword::Items.into(), items::compile))
213        }
214        (_, "maxItems") if ctx.has_vocabulary(&Vocabulary::Validation) => {
215            Some((BuiltinKeyword::MaxItems.into(), max_items::compile))
216        }
217        (_, "maxLength") if ctx.has_vocabulary(&Vocabulary::Validation) => {
218            Some((BuiltinKeyword::MaxLength.into(), max_length::compile))
219        }
220        (_, "maxProperties") if ctx.has_vocabulary(&Vocabulary::Validation) => Some((
221            BuiltinKeyword::MaxProperties.into(),
222            max_properties::compile,
223        )),
224        (_, "minItems") if ctx.has_vocabulary(&Vocabulary::Validation) => {
225            Some((BuiltinKeyword::MinItems.into(), min_items::compile))
226        }
227        (_, "minLength") if ctx.has_vocabulary(&Vocabulary::Validation) => {
228            Some((BuiltinKeyword::MinLength.into(), min_length::compile))
229        }
230        (_, "minProperties") if ctx.has_vocabulary(&Vocabulary::Validation) => Some((
231            BuiltinKeyword::MinProperties.into(),
232            min_properties::compile,
233        )),
234        (_, "multipleOf") if ctx.has_vocabulary(&Vocabulary::Validation) => {
235            Some((BuiltinKeyword::MultipleOf.into(), multiple_of::compile))
236        }
237        (_, "not") if ctx.has_vocabulary(&Vocabulary::Applicator) => {
238            Some((BuiltinKeyword::Not.into(), not::compile))
239        }
240        (_, "oneOf") if ctx.has_vocabulary(&Vocabulary::Applicator) => {
241            Some((BuiltinKeyword::OneOf.into(), one_of::compile))
242        }
243        (_, "pattern") if ctx.has_vocabulary(&Vocabulary::Validation) => {
244            Some((BuiltinKeyword::Pattern.into(), pattern::compile))
245        }
246        (_, "patternProperties") if ctx.has_vocabulary(&Vocabulary::Applicator) => Some((
247            BuiltinKeyword::PatternProperties.into(),
248            pattern_properties::compile,
249        )),
250        (_, "properties") if ctx.has_vocabulary(&Vocabulary::Applicator) => {
251            Some((BuiltinKeyword::Properties.into(), properties::compile))
252        }
253        (_, "required") if ctx.has_vocabulary(&Vocabulary::Validation) => {
254            Some((BuiltinKeyword::Required.into(), required::compile))
255        }
256        (_, "uniqueItems") if ctx.has_vocabulary(&Vocabulary::Validation) => {
257            Some((BuiltinKeyword::UniqueItems.into(), unique_items::compile))
258        }
259        // Draft 4 specific
260        (Draft::Draft4, "maximum") => Some((
261            BuiltinKeyword::Maximum.into(),
262            legacy::maximum_draft_4::compile,
263        )),
264        (Draft::Draft4, "minimum") => Some((
265            BuiltinKeyword::Minimum.into(),
266            legacy::minimum_draft_4::compile,
267        )),
268        (Draft::Draft4, "type") => {
269            Some((BuiltinKeyword::Type.into(), legacy::type_draft_4::compile))
270        }
271        // Draft 6 and later
272        (Draft::Draft6 | Draft::Draft7 | Draft::Draft201909 | Draft::Draft202012, "const")
273            if ctx.has_vocabulary(&Vocabulary::Validation) =>
274        {
275            Some((BuiltinKeyword::Const.into(), const_::compile))
276        }
277        (Draft::Draft6 | Draft::Draft7 | Draft::Draft201909 | Draft::Draft202012, "contains")
278            if ctx.has_vocabulary(&Vocabulary::Applicator) =>
279        {
280            Some((BuiltinKeyword::Contains.into(), contains::compile))
281        }
282        (
283            Draft::Draft6 | Draft::Draft7 | Draft::Draft201909 | Draft::Draft202012,
284            "exclusiveMaximum",
285        ) if ctx.has_vocabulary(&Vocabulary::Validation) => Some((
286            BuiltinKeyword::ExclusiveMaximum.into(),
287            exclusive_maximum::compile,
288        )),
289        (
290            Draft::Draft6 | Draft::Draft7 | Draft::Draft201909 | Draft::Draft202012,
291            "exclusiveMinimum",
292        ) if ctx.has_vocabulary(&Vocabulary::Validation) => Some((
293            BuiltinKeyword::ExclusiveMinimum.into(),
294            exclusive_minimum::compile,
295        )),
296        (Draft::Draft6 | Draft::Draft7 | Draft::Draft201909 | Draft::Draft202012, "maximum")
297            if ctx.has_vocabulary(&Vocabulary::Validation) =>
298        {
299            Some((BuiltinKeyword::Maximum.into(), maximum::compile))
300        }
301        (Draft::Draft6 | Draft::Draft7 | Draft::Draft201909 | Draft::Draft202012, "minimum")
302            if ctx.has_vocabulary(&Vocabulary::Validation) =>
303        {
304            Some((BuiltinKeyword::Minimum.into(), minimum::compile))
305        }
306        (
307            Draft::Draft6 | Draft::Draft7 | Draft::Draft201909 | Draft::Draft202012,
308            "propertyNames",
309        ) if ctx.has_vocabulary(&Vocabulary::Applicator) => Some((
310            BuiltinKeyword::PropertyNames.into(),
311            property_names::compile,
312        )),
313        (Draft::Draft6 | Draft::Draft7 | Draft::Draft201909 | Draft::Draft202012, "type")
314            if ctx.has_vocabulary(&Vocabulary::Validation) =>
315        {
316            Some((BuiltinKeyword::Type.into(), type_::compile))
317        }
318        (Draft::Draft6 | Draft::Draft7, "contentMediaType") => Some((
319            BuiltinKeyword::ContentMediaType.into(),
320            content::compile_media_type,
321        )),
322        (Draft::Draft6 | Draft::Draft7, "contentEncoding") => Some((
323            BuiltinKeyword::ContentEncoding.into(),
324            content::compile_content_encoding,
325        )),
326        (Draft::Draft7 | Draft::Draft201909 | Draft::Draft202012, "if")
327            if ctx.has_vocabulary(&Vocabulary::Applicator) =>
328        {
329            Some((BuiltinKeyword::If.into(), if_::compile))
330        }
331        // Draft 2019-09 specific
332        (Draft::Draft201909, "$recursiveRef") => Some((
333            BuiltinKeyword::RecursiveRef.into(),
334            ref_::compile_recursive_ref,
335        )),
336        (Draft::Draft201909 | Draft::Draft202012, "dependentRequired")
337            if ctx.has_vocabulary(&Vocabulary::Validation) =>
338        {
339            Some((
340                BuiltinKeyword::DependentRequired.into(),
341                dependencies::compile_dependent_required,
342            ))
343        }
344        (Draft::Draft201909 | Draft::Draft202012, "dependentSchemas")
345            if ctx.has_vocabulary(&Vocabulary::Applicator) =>
346        {
347            Some((
348                BuiltinKeyword::DependentSchemas.into(),
349                dependencies::compile_dependent_schemas,
350            ))
351        }
352        (Draft::Draft201909, "unevaluatedItems") if ctx.has_vocabulary(&Vocabulary::Applicator) => {
353            Some((
354                BuiltinKeyword::UnevaluatedItems.into(),
355                unevaluated_items::compile,
356            ))
357        }
358        (Draft::Draft202012, "unevaluatedItems")
359            if ctx.has_vocabulary(&Vocabulary::Unevaluated) =>
360        {
361            Some((
362                BuiltinKeyword::UnevaluatedItems.into(),
363                unevaluated_items::compile,
364            ))
365        }
366        (Draft::Draft201909, "unevaluatedProperties")
367            if ctx.has_vocabulary(&Vocabulary::Applicator) =>
368        {
369            Some((
370                BuiltinKeyword::UnevaluatedProperties.into(),
371                unevaluated_properties::compile,
372            ))
373        }
374        (Draft::Draft202012, "unevaluatedProperties")
375            if ctx.has_vocabulary(&Vocabulary::Unevaluated) =>
376        {
377            Some((
378                BuiltinKeyword::UnevaluatedProperties.into(),
379                unevaluated_properties::compile,
380            ))
381        }
382        // Draft 2020-12 specific
383        (Draft::Draft202012, "prefixItems") if ctx.has_vocabulary(&Vocabulary::Applicator) => {
384            Some((BuiltinKeyword::PrefixItems.into(), prefix_items::compile))
385        }
386        (Draft::Draft202012, "$dynamicRef") => {
387            Some((BuiltinKeyword::DynamicRef.into(), ref_::compile_dynamic_ref))
388        }
389        // Unknown or not-yet-implemented keyword
390        _ => None,
391    }
